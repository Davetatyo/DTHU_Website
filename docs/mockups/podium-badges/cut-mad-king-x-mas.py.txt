# Cuts the podium badges out of the owner's preview image: transparent background (glow and
# lightning keyed against the dark background), an exact transparent hole for the profile
# picture, square PNGs centred on the hole.
import sys, math
from collections import deque
from PIL import Image, ImageFilter
src, out = sys.argv[1], sys.argv[2]
im = Image.open(src).convert("RGB"); P = im.load(); W, H = im.size
BG = tuple(int(v) for v in sys.argv[3].split(','))

def dist(c): return max(abs(c[k] - BG[k]) for k in range(3))

def cut(name, cx, cy, half, hole, keep_from, size, fade_from=None, xlim=None, ylim=None, body=10**6, ymin=0, disc=None, swirl=False, strict=False):
    x0, y0 = int(round(cx - half)), int(round(cy - half)); n = 2 * int(half)
    box = im.crop((x0, y0, x0 + n, y0 + n)); b = box.load()
    lo, hi = xlim if xlim else (0, W)
    # 1. solid silhouette: everything not reachable from the crop border through background
    # Dim glow counts as background too, so the gold halo is keyed (see-through on white)
    # instead of becoming a solid brown rim.
    # The deep red sashes are dark too, but they're part of the badge: strongly red pixels
    # never count as background.
    def dim(c):
        return (0.3 * c[0] + 0.59 * c[1] + 0.11 * c[2]) < 60 and not (c[0] > 45 and c[0] > 1.8 * c[1])
    # Sets with dark metal (Mad King) use the plain colour distance, or the dark plates would
    # be flooded as background.
    bgish = [[(dist(b[x, y]) <= 12) if strict else dim(b[x, y]) for x in range(n)] for y in range(n)]
    seen = [[False] * n for _ in range(n)]; q = deque()
    for i in range(n):
        for (x, y) in ((i, 0), (i, n - 1), (0, i), (n - 1, i)):
            if bgish[y][x] and not seen[y][x]: seen[y][x] = True; q.append((x, y))
    while q:
        x, y = q.popleft()
        for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
            if 0 <= nx < n and 0 <= ny < n and not seen[ny][nx] and bgish[ny][nx]:
                seen[ny][nx] = True; q.append((nx, ny))
    solid = Image.new("L", (n, n), 0); sp = solid.load()
    for y in range(n):
        for x in range(n):
            if not seen[y][x]: sp[x, y] = 255
    solid = solid.filter(ImageFilter.MaxFilter(5))  # give back the dark outline eaten by the fill
    # Dark pockets enclosed by lightning outside the ring body aren't part of the badge: keep
    # the solid area only inside the body radius or right next to coloured pixels.
    bright = Image.new("L", (n, n), 0); bp = bright.load()
    for y in range(n):
        for x in range(n):
            if dist(b[x, y]) > 30: bp[x, y] = 255
    bright = bright.filter(ImageFilter.MaxFilter(7)); bp = bright.load()
    sp = solid.load()
    for y in range(n):
        for x in range(n):
            if sp[x, y] and math.hypot(x + 0.5 - (cx - x0), y + 0.5 - (cy - y0)) > body and not bp[x, y]: sp[x, y] = 0
    res = Image.new("RGBA", (n, n)); r = res.load()
    for y in range(n):
        for x in range(n):
            c = b[x, y]; gx, gy = x0 + x, y0 + y
            rr = math.hypot(x + 0.5 - (cx - x0), y + 0.5 - (cy - y0))
            if gx < lo or gx >= hi or (ylim and gy >= ylim) or gy < ymin: r[x, y] = (0, 0, 0, 0); continue
            if sp[x, y]: a = 1.0
            else:  # glow / lightning: key against the background colour
                # only light (glow, lightning, sparks) survives; dark shadow pixels drop out, so
                # nothing smudges on the white profile card
                lum = 0.3 * c[0] + 0.59 * c[1] + 0.11 * c[2]
                a = 1.0 if (c[0] > 45 and c[0] > 1.8 * c[1]) else max(0.0, min(1.0, (lum - 22) / 80))
            # the hole: transparent inside `hole`, except bright ornaments overlapping it from `keep_from`
            if rr < keep_from: a = 0.0
            elif rr < hole:
                # the dark band and the grey stand-in picture go; ornaments overlapping the hole stay
                if dist(c) < 45 or (disc and all(abs(c[k] - disc[k]) <= 16 for k in range(3))) or (swirl and c[0] > 1.35 * c[1] and c[0] > 1.3 * c[2] and 0.3 * c[0] + 0.59 * c[1] + 0.11 * c[2] < 120): a = 0.0
            if rr < hole + 1 and rr >= hole and dist(c) < 45: a *= (rr - hole)  # soft edge
            if fade_from and rr > fade_from and not sp[x, y]: a *= max(0.0, 1 - (rr - fade_from) / (half - fade_from))
            if a <= 0: r[x, y] = (0, 0, 0, 0); continue
            if a < 1:  # un-premultiply the background out of the glow
                col = tuple(max(0, min(255, int(round((c[k] - BG[k] * (1 - a)) / a)))) for k in range(3))
            else: col = c
            r[x, y] = (*col, int(round(a * 255)))
    res = res.resize((size, size), Image.LANCZOS)
    res.save(f"{out}/{name}.png", optimize=True)
    return hole * size / n

import json
cfg = json.loads(sys.argv[4])
for k, c in cfg["large"].items():
    h = cut(f"{k}-512", c[0], c[1], cfg["L"]["half"], cfg["L"]["hole"], cfg["L"]["keep"], 512, fade_from=cfg["L"]["fade"], xlim=tuple(c[2]), ylim=cfg["L"]["ylim"], body=cfg["L"]["body"], ymin=cfg["L"].get("ymin", 0), disc=tuple(cfg["disc"]), swirl=cfg.get("swirl", False), strict=cfg.get("strict", False))
    print(k, "large hole radius in 512 canvas:", round(h, 2))
for k, c in cfg["small"].items():
    hs = cfg["S"]["half"]
    h = cut(f"{k}-128", c[0], c[1], hs, cfg["S"]["hole"], cfg["S"]["keep"], 128, xlim=(c[0] - hs, c[0] + hs), ylim=cfg["S"]["ylim"], ymin=cfg["S"]["ymin"], disc=tuple(cfg["disc"]), swirl=cfg.get("swirl", False), strict=cfg.get("strict", False))
    print(k, "small hole radius in 128 canvas:", round(h, 2))
