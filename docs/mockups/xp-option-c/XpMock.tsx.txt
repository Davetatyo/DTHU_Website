// MOCKUP ONLY (scratch copy): option C. Badge ring on the avatar, XP as a bar under the name.
import "./xpmock.css";

export const MOCK = { level: 5, xp: 172, from: 140, to: 200, avatar: "/szabalytar/img/heroes/barbarian.webp" };
export const pct = ((MOCK.xp - MOCK.from) / (MOCK.to - MOCK.from)) * 100;

function studs(c: number, r: number, n: number, k: number) {
  return Array.from({ length: n }, (_, i) => {
    const a = (i / n) * 2 * Math.PI;
    const x = c + r * Math.sin(a);
    const y = c - r * Math.cos(a);
    return i === 0 ? (
      <path key={i} d={`M${x} ${y - k * 2.2} L${x + k * 1.6} ${y} L${x} ${y + k * 2.2} L${x - k * 1.6} ${y} Z`} fill="#f2c95a" stroke="#6b430a" strokeWidth={0.8} />
    ) : (
      <circle key={i} cx={x} cy={y} r={k} fill="#f2c95a" />
    );
  });
}

/** Avatar with the worn badge ring ("Évadbajnok" example). */
export function BadgeAvatar({ size, ring, studCount }: { size: number; ring: number; studCount: number }) {
  const c = size / 2;
  const r = c - ring / 2 - 1;
  return (
    <span className="xm-av" style={{ width: size, height: size }}>
      <span className="xm-img" style={{ inset: ring + 1, backgroundImage: `url(${MOCK.avatar})` }} />
      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden>
        <circle cx={c} cy={c} r={r} fill="none" stroke="#5b2d9e" strokeWidth={ring} />
        <circle cx={c} cy={c} r={r + ring / 2 - 0.6} fill="none" stroke="#d9a83f" strokeWidth={1.2} />
        <circle cx={c} cy={c} r={r - ring / 2 + 0.6} fill="none" stroke="#d9a83f" strokeWidth={1.2} />
        {studs(c, r, studCount, Math.max(1.4, ring * 0.3))}
      </svg>
    </span>
  );
}

/** Header chip content: badge avatar, name and the thin XP bar under it. */
export function ChipMock({ name }: { name: string }) {
  return (
    <>
      <BadgeAvatar size={28} ring={3.4} studCount={4} />
      <span className="xm-name">
        <span className="truncate">{name}</span>
        <span className="xm-bar" title={`${MOCK.level}. szint · ${MOCK.xp} / ${MOCK.to} XP`}>
          <span style={{ width: `${pct}%` }} />
        </span>
      </span>
    </>
  );
}
